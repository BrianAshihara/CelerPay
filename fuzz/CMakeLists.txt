function(celerpay_add_fuzzer name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE celerpay::celerpay)
    if(MSVC)
        target_compile_options(${name} PRIVATE /fsanitize=fuzzer)
        target_link_options(${name} PRIVATE /INCREMENTAL:NO)
    else()
        target_compile_options(${name} PRIVATE -fsanitize=fuzzer)
        target_link_options(${name} PRIVATE -fsanitize=fuzzer)
    endif()
endfunction()

celerpay_add_fuzzer(tlv_fuzz)
celerpay_add_fuzzer(pix_fuzz)
celerpay_add_fuzzer(boleto_fuzz)
celerpay_add_fuzzer(statement_fuzz)
